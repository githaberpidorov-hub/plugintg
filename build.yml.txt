name: Build Plugin

on:
  push:
    branches: [ "main" ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
    - name: Checkout
      uses: actions/checkout@v4

    - name: Package Plugin
      run: |
        mkdir -p output
        cp metadata.json output/
        cp loader.py output/
        cd output
        zip auygram_enhancer.plugin metadata.json loader.py

    - name: Upload .plugin artifact
      uses: actions/upload-artifact@v4
      with:
        name: auygram-plugin
        path: output/auygram_enhancer.plugin
